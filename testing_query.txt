#raw_event.ip  match raw_ip_location.ip
bq query --use_legacy_sql=false "
SELECT
    COUNT(*) AS total_event_rows_with_ip
    , COUNTIF(loc.ip IS NULL) AS unmatched_ip_rows
    , COUNTIF(loc.ip IS NOT NULL) AS matched_ip_rows
FROM \`project-6978f7f5-636f-40bd-83c.glamira_dataset_raw.raw_event\` AS event
LEFT JOIN \`project-6978f7f5-636f-40bd-83c.glamira_dataset_raw.raw_ip_location\` AS loc
    ON event.ip = loc.ip
WHERE event.ip IS NOT NULL
"
