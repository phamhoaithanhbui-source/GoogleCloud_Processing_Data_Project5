#Check raw data
bq query --use_legacy_sql=false "
SELECT
  COUNT(*) AS total_rows,
  COUNT(DISTINCT product_id) AS distinct_product_id,
  COUNTIF(product_id IS NULL) AS null_product_id,
  COUNTIF(product_name IS NULL) AS null_product_name
FROM \`project-6978f7f5-636f-40bd-83c.glamira_dataset.raw_product\`



# Check duplicate product
bq query --use_legacy_sql=false "
SELECT product_id, COUNT(*) AS cnt
FROM \`project-6978f7f5-636f-40bd-83c.glamira_dataset.raw_product\`
GROUP BY product_id
HAVING cnt > 1
ORDER BY cnt DESC
LIMIT 20
"
